import { describe, expect, it } from 'vitest'

import { readDocx } from '../src/main/office-text.js'

/*
 * OT1. A self-closing <w:t/> is matched by the regex's first alternative as
 * an opening tag, so raw XML up to the next </w:t> becomes the paragraph's
 * words.
 */
function storedZip(files: Record<string, string>): Buffer {
  const locals: Buffer[] = []
  const centrals: Buffer[] = []
  let offset = 0
  let count = 0
  for (const [name, text] of Object.entries(files)) {
    const nameBuf = Buffer.from(name, 'utf8')
    const data = Buffer.from(text, 'utf8')
    const local = Buffer.alloc(30)
    local.writeUInt32LE(0x04034b50, 0)
    local.writeUInt16LE(20, 4)
    local.writeUInt32LE(data.length, 18)
    local.writeUInt32LE(data.length, 22)
    local.writeUInt16LE(nameBuf.length, 26)
    const central = Buffer.alloc(46)
    central.writeUInt32LE(0x02014b50, 0)
    central.writeUInt16LE(20, 4)
    central.writeUInt16LE(20, 6)
    central.writeUInt32LE(data.length, 20)
    central.writeUInt32LE(data.length, 24)
    central.writeUInt16LE(nameBuf.length, 28)
    central.writeUInt32LE(offset, 42)
    locals.push(local, nameBuf, data)
    centrals.push(central, nameBuf)
    offset += 30 + nameBuf.length + data.length
    count += 1
  }
  const centralSize = centrals.reduce((sum, part) => sum + part.length, 0)
  const end = Buffer.alloc(22)
  end.writeUInt32LE(0x06054b50, 0)
  end.writeUInt16LE(count, 8)
  end.writeUInt16LE(count, 10)
  end.writeUInt32LE(centralSize, 12)
  end.writeUInt32LE(offset, 16)
  return Buffer.concat([...locals, ...centrals, end])
}

describe('OT1: a Word paragraph with an empty self-closing run', () => {
  it('reads as its words alone', () => {
    const xml = '<?xml version="1.0"?><w:document xmlns:w="w"><w:body><w:p><w:r><w:t/></w:r><w:r><w:t>Hello</w:t></w:r></w:p></w:body></w:document>'
    const document = readDocx(storedZip({ '[Content_Types].xml': '<Types/>', 'word/document.xml': xml }))
    const paragraphs = document.blocks.flatMap((block) => (block.kind === 'paragraph' ? [block.text] : []))
    expect(paragraphs).toEqual(['Hello'])
  })
})
